## 1. 一键全量侦察（推荐）

**作用**：一次跑完，拿到课程结构、目录、任务点、iframe、视频、全局函数、隐藏 input、storage、cookies、上报 URL 的全部信息。

**用法**：打开课程页 → F12 → Console → 整段粘贴 → 回车。结果会自动复制到剪贴板，并挂载到 `window.__reconData`。

```js
(async function ultraRecon() {
  'use strict';

  const lines = [];
  const push = (s = '') => lines.push(s);
  const trunc = (s, n = 200) => {
    if (s == null) return '';
    s = String(s).replace(/\s+/g, ' ').trim();
    return s.length > n ? s.slice(0, n) + '…' : s;
  };
  const isVisible = (el) => {
    if (!el || !el.getBoundingClientRect) return false;
    const r = el.getBoundingClientRect();
    if (r.width <= 0 || r.height <= 0) return false;
    const st = getComputedStyle(el);
    return st.display !== 'none' && st.visibility !== 'hidden' && st.opacity !== '0';
  };

  const data = {
    meta: {},
    catalog: { chapters: [], sections: [], tree: [], unfinished: [], stats: {} },
    tabs: [],
    globals: { functions: {}, values: {}, interesting: [] },
    frames: [],
    taskPoints: [],
    videos: [],
    inputs: { hidden: [], visible: [] },
    storage: { local: {}, session: {} },
    cookies: [],
    popups: [],
    buttons: [],
    scripts: [],
    reportUrls: []
  };

  // ===== 1. Meta =====
  data.meta = {
    url: location.href,
    host: location.host,
    pathname: location.pathname,
    searchParams: Object.fromEntries(new URLSearchParams(location.search).entries()),
    hash: location.hash,
    title: document.title,
    time: new Date().toLocaleString('zh-CN'),
    isoTime: new Date().toISOString(),
    isTop: window === window.top,
    referrer: document.referrer,
    readyState: document.readyState,
    ua: navigator.userAgent
  };

  // ===== 2. 目录树 =====
  (function scanCatalog() {
    let chapter = null;
    document.querySelectorAll('.posCatalog_select').forEach((node) => {
      const isChapter = node.classList.contains('firstLayer');
      const id = (node.id || '').replace(/^cur/, '');
      const label = node.querySelector('.posCatalog_sbar')?.textContent?.trim() || '';
      const nameEl = node.querySelector('.posCatalog_name') || node.querySelector('.posCatalog_title');
      const name = nameEl?.getAttribute('title') || nameEl?.textContent?.trim() || '';
      const finished = !!node.querySelector('.icon_Completed');
      const active = node.classList.contains('active') || node.classList.contains('cur');
      const rec = { id, label, name, finished, active, isChapter };
      if (isChapter) {
        data.catalog.chapters.push(rec);
        chapter = rec;
      } else {
        data.catalog.sections.push(rec);
        if (!finished) data.catalog.unfinished.push(rec);
        data.catalog.tree.push({
          chapter: chapter ? { id: chapter.id, label: chapter.label, name: chapter.name } : null,
          section: { id: rec.id, label: rec.label, name: rec.name, finished: rec.finished, active: rec.active }
        });
      }
    });
    data.catalog.stats = {
      chapters: data.catalog.chapters.length,
      sections: data.catalog.sections.length,
      finished: data.catalog.sections.filter(s => s.finished).length,
      unfinished: data.catalog.unfinished.length,
      activeChapter: data.catalog.chapters.find(c => c.active) || null,
      activeSection: data.catalog.sections.find(s => s.active) || null
    };
  })();

  // ===== 3. 标签栏 =====
  document.querySelectorAll('#prev_tab .prev_ul > li, .prev_list li').forEach((t) => {
    data.tabs.push({
      id: t.id || '',
      className: String(t.className || ''),
      cardid: t.getAttribute('cardid') || '',
      active: t.classList.contains('active'),
      text: trunc(t.querySelector('.spanText')?.textContent || t.textContent, 40)
    });
  });

  // ===== 4. 全局函数与值 =====
  const NAMES = [
    'checkJob','playFirstVideo','getChapterNextVideo','getTeacherAjax',
    'changeDisplayContent','videoAutoPlay','PCount','getJob','setJob',
    'toNext','toPrev','nextChapter','prevChapter','loadChapter',
    'jumpTo','autoPlay','playVideo','pauseVideo','switchChapter',
    'getNextChapter','getPrevChapter'
  ];
  NAMES.forEach((n) => {
    try {
      const v = window[n];
      const t = typeof v;
      if (t === 'undefined') { data.globals.functions[n] = { exists: false }; return; }
      if (t === 'function') {
        let body = '';
        try { body = v.toString().slice(0, 2000); } catch (e) { body = '(取源码失败)'; }
        data.globals.functions[n] = { exists: true, type: 'function', body };
      } else if (t === 'string' || t === 'number' || t === 'boolean') {
        data.globals.values[n] = { type: t, value: v };
      } else if (t === 'object') {
        let keys = [];
        try { keys = Object.keys(v).slice(0, 50); } catch (e) {}
        let preview = '';
        try { preview = JSON.stringify(v); } catch (e) { preview = '(循环引用)'; }
        data.globals.values[n] = { type: 'object', keys, preview: trunc(preview, 800) };
      }
    } catch (e) {
      data.globals.functions[n] = { exists: false, error: String(e) };
    }
  });

  try {
    data.globals.interesting = Object.keys(window)
      .filter(k => !/^[a-z]$/.test(k) && /job|video|chapter|play|study|course|progress|task|point|next|prev|player/i.test(k))
      .slice(0, 80)
      .map(k => {
        let t = 'error';
        try { t = typeof window[k]; } catch (e) {}
        return { key: k, type: t };
      });
  } catch (e) {}

  // ===== 5. 递归扫描 iframe =====
  const visited = new Set();
  const frameList = [];

  function scanDoc(doc, path, depth) {
    if (!doc || visited.has(doc) || depth > 8) return;
    visited.add(doc);
    let docUrl = '', docTitle = '';
    try { docUrl = doc.location.href; } catch (e) { docUrl = '(跨域)'; }
    try { docTitle = doc.title; } catch (e) {}

    const rec = {
      path, depth, url: docUrl, title: docTitle,
      crossOrigin: false,
      taskPointIcons: 0, attachCts: 0, finishedCt: 0,
      hasNext: false, hasPrev: false,
      jobs: [], videos: [], childFrames: []
    };

    const attachCts = doc.querySelectorAll('.ans-attach-ct');
    rec.taskPointIcons = doc.querySelectorAll('.ans-job-icon').length;
    rec.attachCts = attachCts.length;
    rec.finishedCt = doc.querySelectorAll('.ans-job-finished').length;
    rec.hasNext = !!doc.querySelector('.next, #prevNextFocusNext');
    rec.hasPrev = !!doc.querySelector('.prev, #prevNextFocusPrev');

    attachCts.forEach((c, i) => {
      const icon = c.querySelector('.ans-job-icon');
      const done = c.classList.contains('ans-job-finished') || !!c.querySelector('.ans-job-finished');
      const aria = icon?.getAttribute('aria-label') || '';
      const iconClass = String(icon?.className || '');
      const iconData = icon?.getAttribute('data') || '';
      const jobId = c.getAttribute('jobid') || c.getAttribute('data-jobid')
                 || c.querySelector('[jobid]')?.getAttribute('jobid') || '';
      const objectId = c.getAttribute('objectid') || c.getAttribute('data-objectid') || '';
      const type = c.getAttribute('type') || c.getAttribute('data-type') || '';
      const titleEl = c.querySelector('.ans-job-title, .attach-title, [class*="title"], .ans_attach_title, .ans-job-name');
      const title = titleEl?.textContent?.trim() || c.getAttribute('title') || '';
      const videoEl = c.querySelector('video');
      const iframeEl = c.querySelector('iframe');
      const src = iframeEl ? iframeEl.src : '';

      let inferredType = type;
      if (!inferredType) {
        if (videoEl || /video|ananas\/modules\/video/.test(src)) inferredType = 'video';
        else if (/pdf|doc|document|ananas\/modules\/doc/.test(src)) inferredType = 'document';
        else if (/quiz|test|exam/.test(src) || /测验|测试/.test(title)) inferredType = 'quiz';
        else if (/discuss|bbs/.test(src) || /讨论/.test(title)) inferredType = 'discuss';
        else inferredType = 'unknown';
      }

      const job = {
        index: i, done, type: inferredType, title: trunc(title, 60),
        aria, iconClass: trunc(iconClass, 100), iconData,
        jobId, objectId,
        hasVideo: !!videoEl,
        iframeSrc: trunc(src, 200),
        cardid: c.getAttribute('cardid') || '',
        className: trunc(String(c.className || ''), 100)
      };
      rec.jobs.push(job);
      data.taskPoints.push({ framePath: path, ...job });
    });

    doc.querySelectorAll('video').forEach((v, i) => {
      const vrec = {
        framePath: path, index: i,
        src: trunc(v.currentSrc || v.src, 200),
        duration: +(v.duration || 0).toFixed(2),
        currentTime: +(v.currentTime || 0).toFixed(2),
        paused: v.paused, ended: v.ended,
        muted: v.muted, volume: v.volume,
        playbackRate: v.playbackRate,
        readyState: v.readyState,
        networkState: v.networkState,
        buffered: (() => {
          try {
            const b = v.buffered, ranges = [];
            for (let j = 0; j < b.length; j++) ranges.push([+b.start(j).toFixed(2), +b.end(j).toFixed(2)]);
            return ranges;
          } catch (e) { return []; }
        })(),
        attributes: (() => {
          const a = {};
          for (const attr of v.attributes) a[attr.name] = trunc(attr.value, 100);
          return a;
        })()
      };
      rec.videos.push(vrec);
      data.videos.push(vrec);
    });

    frameList.push(rec);

    doc.querySelectorAll('iframe').forEach((f, i) => {
      const childPath = path + ' > iframe[' + i + ']';
      let childDoc = null;
      try { childDoc = f.contentDocument; } catch (e) { childDoc = null; }
      if (childDoc) {
        scanDoc(childDoc, childPath, depth + 1);
      } else {
        rec.childFrames.push({ path: childPath, src: trunc(f.src, 200), crossOrigin: true });
        frameList.push({
          path: childPath, depth: depth + 1, url: trunc(f.src, 200),
          crossOrigin: true,
          taskPointIcons: 0, attachCts: 0, finishedCt: 0,
          hasNext: false, hasPrev: false,
          jobs: [], videos: [], childFrames: []
        });
      }
    });
  }
  scanDoc(document, 'top', 0);
  data.frames = frameList;

  // ===== 6. input =====
  document.querySelectorAll('input').forEach((inp) => {
    const type = inp.type || 'text';
    const rec = { id: inp.id || '', name: inp.name || '', type, value: trunc(inp.value, 200) };
    if (type === 'hidden') {
      if (inp.id || inp.name) data.inputs.hidden.push(rec);
    } else if (isVisible(inp)) {
      data.inputs.visible.push(rec);
    }
  });

  // ===== 7. storage =====
  const reKey = /course|chapter|clazz|user|token|job|video|study|card|knowledge|progress|point|task/i;
  try {
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (reKey.test(k) || localStorage.length < 30) data.storage.local[k] = trunc(localStorage.getItem(k), 200);
    }
  } catch (e) {}
  try {
    for (let i = 0; i < sessionStorage.length; i++) {
      const k = sessionStorage.key(i);
      if (reKey.test(k) || sessionStorage.length < 30) data.storage.session[k] = trunc(sessionStorage.getItem(k), 200);
    }
  } catch (e) {}

  // ===== 8. cookies =====
  try {
    document.cookie.split(';').forEach((c) => {
      const [k, ...v] = c.split('=');
      if (reKey.test(k.trim())) data.cookies.push({ name: k.trim(), value: trunc(v.join('=').trim(), 100) });
    });
  } catch (e) {}

  // ===== 9. 弹窗 =====
  document.querySelectorAll('[class*="mask"],[class*="pop"],[class*="dialog"],[class*="modal"],[class*="Alert"],[class*="layer"],[class*="tips"]').forEach((p) => {
    const st = getComputedStyle(p);
    if (st.display === 'none' || st.visibility === 'hidden' || !isVisible(p)) return;
    data.popups.push({
      tag: p.tagName.toLowerCase(), id: p.id || '',
      className: trunc(String(p.className || ''), 100),
      text: trunc(p.textContent, 120)
    });
  });

  // ===== 10. 按钮 =====
  const KEY = ['下一节','下一章','上一节','上一章','去学习','去完成','提交','确定','确认','取消','继续','关闭','开始','返回','播放','暂停','保存','重播','完成'];
  const seen = new Set();
  document.querySelectorAll('button, a, [role="button"], input[type="button"], input[type="submit"], .btn, [class*="btn"]').forEach((b) => {
    if (!isVisible(b)) return;
    const text = (b.textContent || b.value || '').trim();
    if (!text || !KEY.some(k => text.includes(k))) return;
    const key = text + '|' + b.tagName + '|' + b.className;
    if (seen.has(key)) return;
    seen.add(key);
    data.buttons.push({
      text: trunc(text, 30), tag: b.tagName.toLowerCase(),
      id: b.id || '', className: trunc(String(b.className || ''), 100),
      href: trunc(b.getAttribute('href') || '', 200)
    });
  });

  // ===== 11. 可疑上报 URL =====
  const reUrl = /report|progress|upload|heartbeat|log|record/i;
  document.querySelectorAll('script[src], link[href]').forEach((el) => {
    const url = el.src || el.href || '';
    if (reUrl.test(url)) data.reportUrls.push(trunc(url, 200));
  });

  // ===== 12. scripts =====
  document.querySelectorAll('script[src]').forEach((s) => data.scripts.push(trunc(s.src, 200)));

  // ===== 输出文本 =====
  push('╔══════════════════════════════════════════════════════════╗');
  push('║  超星/学习通 全量侦察报告 v3.0 Ultra                     ║');
  push('╚══════════════════════════════════════════════════════════╝');
  push(`时间   : ${data.meta.time}`);
  push(`URL    : ${data.meta.url}`);
  push(`标题   : ${data.meta.title}`);
  push(`顶层   : ${data.meta.isTop}`);
  push(`Ready  : ${data.meta.readyState}`);
  push(`参数   : ${JSON.stringify(data.meta.searchParams)}`);
  push('');

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【1. 目录结构】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push(`章 ${data.catalog.stats.chapters} | 节 ${data.catalog.stats.sections} | 已完成 ${data.catalog.stats.finished} | 未完成 ${data.catalog.stats.unfinished}`);
  if (data.catalog.stats.activeSection) push(`当前节: [${data.catalog.stats.activeSection.label}] ${data.catalog.stats.activeSection.name} (id=${data.catalog.stats.activeSection.id})`);
  push('');
  push('章节树:');
  data.catalog.tree.forEach((n) => {
    const c = n.chapter || {};
    const s = n.section || {};
    push(`  [${s.label}] ${s.finished ? '✓' : '✗'}${s.active ? '★' : ' '} ${s.name}  (id=${s.id}, 章: ${c.label} ${c.name})`);
  });
  push('');
  push('未完成小节:');
  data.catalog.unfinished.forEach((s) => push(`  ✗ [${s.label}] ${s.name}  (id=${s.id})`));
  push('');

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【2. 标签栏 / 卡片】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push(`数量: ${data.tabs.length}`);
  data.tabs.forEach((t, i) => push(`  [${i+1}] id=${t.id} active=${t.active} cardid=${t.cardid} "${t.text}"`));
  push('');

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【3. 全局函数 / 值】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('存在性:');
  Object.entries(data.globals.functions).forEach(([k, v]) => {
    push(`  ${v.exists ? '✓' : '✗'} window.${k}${v.exists ? ' → ' + v.type : ''}`);
  });
  push('');
  push('函数源码预览:');
  Object.entries(data.globals.functions).forEach(([k, v]) => {
    if (!v.exists || !v.body) return;
    push(`\n  ── ${k} ──`);
    push(`  ${trunc(v.body, 800)}`);
  });
  push('');
  push('值类型全局对象:');
  Object.entries(data.globals.values).forEach(([k, v]) => {
    push(`  ${k}: type=${v.type}`);
    if (v.value !== undefined) push(`     value = ${trunc(String(v.value), 300)}`);
    if (v.keys) push(`     keys  = ${v.keys.slice(0, 25).join(', ')}`);
    if (v.preview) push(`     preview = ${v.preview}`);
  });
  push('');
  if (data.globals.interesting.length) {
    push('相关全局属性:');
    data.globals.interesting.forEach((g) => push(`  ${g.key}: ${g.type}`));
    push('');
  }

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【4. Frame 递归扫描】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push(`共 ${data.frames.length} 个 frame`);
  data.frames.forEach((f) => {
    push(`\n  ── ${f.path} ${f.crossOrigin ? '(跨域)' : ''} ──`);
    push(`     url: ${f.url}`);
    if (f.title) push(`     title: ${f.title}`);
    if (f.crossOrigin) return;
    push(`     任务点: icons=${f.taskPointIcons} attach=${f.attachCts} finished=${f.finishedCt} next=${f.hasNext} prev=${f.hasPrev}`);
    f.jobs.forEach((j) => {
      push(`       [${j.index}] ${j.done ? '✓' : '✗'} type=${j.type} jobId=${j.jobId} objectId=${j.objectId} cardid=${j.cardid}`);
      if (j.title) push(`           title: ${j.title}`);
      if (j.iframeSrc) push(`           iframe: ${j.iframeSrc}`);
    });
    f.videos.forEach((v, i) => {
      push(`       video[${i}] ${v.currentTime}/${v.duration}s paused=${v.paused} rate=${v.playbackRate} readyState=${v.readyState} buffered=${JSON.stringify(v.buffered)}`);
      push(`           src: ${v.src}`);
    });
  });
  push('');

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【5. 任务点汇总】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push(`总数 ${data.taskPoints.length} | 未完成 ${data.taskPoints.filter(t => !t.done).length}`);
  data.taskPoints.forEach((t) => {
    push(`  ${t.done ? '✓' : '✗'} [${t.framePath}] type=${t.type} jobId=${t.jobId} title="${t.title}"`);
  });
  push('');

  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push('【6. 视频汇总】');
  push('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  push(`总数 ${data.videos.length}`);
  data.videos.forEach((v, i) => {
    push(`  [${i}] [${v.framePath}]`);
    push(`      ${v.currentTime}/${v.duration}s paused=${v.paused} ended=${v.ended} rate=${v.playbackRate} readyState=${v.readyState} netState=${v.networkState}`);

  // 3. 混淆函数字典
  P('\n【4. 混淆函数字典还原】');
  ['_0x2d9b','_0x2b9d41'].forEach(fn => {
    try {
      const f = vWin[fn];
      if (typeof f !== 'function') { P(`  ✗ ${fn} 不存在`); return; }
      P(`  ✓ ${fn} 存在`);
      [0x49b, 0x460, 0x3e6, 0x492, 0x1a0, 0x293, 0x3be, 0x36a, 0x2ae, 0x3cc, 0x493, 0x38b, 0x469, 0x41e, 0x479].forEach(idx => {
        try {
          const r = f(idx);
          P(`    ${fn}(${idx.toString(16)}) = ${String(r).slice(0, 60)}`);
        } catch(e) {}
      });
    } catch(e) {}
  });

  // 5. 播放 60 秒
  P('\n【5. 播放 60 秒，统计上报】');
  const v = vDoc.querySelector('video');
  if (!v) { P('  ✗ 无 video'); return; }

  const start = Date.now();
  const startCT = v.currentTime;
  const startCapCount = captured.length;

  try { await v.play(); } catch(e) { P('  play 失败: '+e.message); }

  for (let i = 0; i < 12; i++) {
    await sleep(5000);
    const elapsed = ((Date.now()-start)/1000).toFixed(1);
    P(`  [${elapsed}s] CT=${v.currentTime.toFixed(2)} 累计上报=${captured.length - startCapCount}`);
  }

  try { v.pause(); } catch(e) {}
  P(`\n  播放结束：从 ${startCT.toFixed(2)} 到 ${v.currentTime.toFixed(2)}`);

  // 6. 汇总
  P('\n【6. 上报汇总】');
  P(`  共 ${captured.length} 条`);
  captured.forEach((r, i) => {
    P(`\n  === [${i}] ${r.type} ${r.method} ===`);
    P(`  URL: ${r.url}`);
    if (r.body) P(`  BODY: ${r.body}`);
  });

  if (captured.length > 1) {
    P('\n  上报间隔:');
    for (let i = 1; i < captured.length; i++) {
      P(`    [${i-1}→${i}] ${captured[i].t - captured[i-1].t} ms`);
    }
  }

  P('\n╔══════════════════════════════════════════════╗');
  P('║  狩猎完成                                      ║');
  P('╚══════════════════════════════════════════════╝');

  window.__reportHunter = { out, captured };
  try { await navigator.clipboard.writeText(out.join('\n')); console.log('✅ 已复制到剪贴板'); } catch(e) { console.log('⚠️ 复制失败'); }
})();
```

## 6. 数据解读

### 关键字段

| 字段                           | 含义                                      |
| ------------------------------ | ----------------------------------------- |
| `catalog.unfinished`           | 未完成的小节列表，含 id（可直接拼 URL）   |
| `taskPoints[].jobId`           | 任务点唯一标识，上报接口用                |
| `taskPoints[].objectId`        | 资源标识，和视频 URL 相关                 |
| `taskPoints[].done`            | 是否已完成（aria + class 双判定）         |
| `videos[].readyState`          | 0=无数据 1=元数据 2=当前帧 3=可播 4=足够  |
| `videos[].networkState`        | 0=空 1=空闲 2=加载中 3=无源               |
| `globals.values.videoAutoPlay` | "0" = 官方自动播放关闭，需手动 play       |
| `inputs.hidden._studystate`    | `{nextChapterId, unfinishCount}` 全局进度 |
| `meta.searchParams.chapterId`  | 当前节 id                                 |
| `meta.searchParams.courseId`   | 当前课程 id                               |

### 拼接跳转 URL

```js
// 从 catalog.unfinished 里取 id，替换 URL 的 chapterId
const url = new URL(location.href);
url.searchParams.set('chapterId', '1198980960');
location.href = url.toString();
```

### 常见问题

**Q: 跨域 iframe 抓不到内容？**  
A: 只读脚本受同源策略限制，只能拿到 `src`。要看内部 DOM 需要装 Edge 扩展。

**Q: 任务点显示 done=true 但超星还是让重播？**  
A: DOM 状态可能滞后。以服务端重扫目录的 `icon_Completed` 为准。

**Q: `metadata timeout` 是什么？**  
A: 视频源还没就绪就调 `play()`。`readyState=0` 时先 `load()` 等元数据。

**Q: 上报 URL 被截断？**  
A: 用"上报机制狩猎"，它不截断。
